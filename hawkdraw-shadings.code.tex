% Shadings

% ===

% PDF treats shadings as patterns and hence as fill (or stroke). But we want to ensure that shadings are
% reusable for different sizes or rotations, thus we apply the approach of PGF/TikZ here.

% Shadings are defined as rectangular PDF Objects. There are various types of shadings, but we only define
% linear, radial and conic shaings here having type 2, 3 and 6 respectively. Following the PGF approach, ee
% define shadings as square with a size of 100bp times 100bp. Type 2 and 3 shadings are defined by functions,
% type 6 shadings are defined by a binary stream.

% In order to allow for rotation of shadings, we use the PGF approach of defining the shadings twice the
% size of the path they should fill. Thus, we need to use multiple functions to define the shading, because
% the outer part of the shading object will be clipped away by the path. We do this by defining the
% functions in a nested way. The outer function is a type 3 function, which defines the inner functions.
% The inner functions depend on the type of the shading and the number of colors used.

% Every shading defined by its type, the used colors and whether or not it has some offset (for radial or
% conic shadings) is defined only once and a PDF XObject is created for it. In the PDF, two XObjects are
% created, one that defines the shading using functions and one that represents a box referring to the
% shading. Upon use, the latter XObject is (re-)used by placing it in a box which is then scaled to twice the
% size of the path bounding box, rotated and placed at the center of the relevant path, then clipped by this
% path.

% Using this approach, it is not possible to apply shadings to strokes. In order to to that, we would need to
% define shadings as patterns and record the current position of the path in order to position the pattern
% correctly. This would require two compilation runs.

% ===

\cs_new_protected:Npn \hawkdraw_shading_fill:n #1 {
    \hawkdraw_shading_create:nV {#1} \l_hawkdraw_shading_colors_clist
    \hbox_set:Nn \l_hawkdraw_shading_box {
        \hawkdraw_shading_use:n {
            #1
            . \clist_use:N \l_hawkdraw_shading_colors_clist
            \str_if_eq:nnT {#1} { radial } {
                . \fp_to_decimal:n { \l_hawkdraw_shading_offset_fp }
            }
            \str_if_eq:nnT {#1} { conic } {
                . \fp_to_decimal:n { \l_hawkdraw_shading_angle_delta_fp }
            }
        }
    }
    % scale to 2 * width of current bbox, 2 * height of current bbox
    % translate to center of current bbox
    \draw_transform_shift:n {
        (
            \l_hawkdraw_shading_bbox_xmin_dim + (
                \l_hawkdraw_shading_bbox_xmax_dim -
                \l_hawkdraw_shading_bbox_xmin_dim
            ) / 2
        ,
            \l_hawkdraw_shading_bbox_ymin_dim + (
                \l_hawkdraw_shading_bbox_ymax_dim -
                \l_hawkdraw_shading_bbox_ymin_dim
            ) / 2
        )
    }
    \draw_transform_rotate:n {
        \l_hawkdraw_shading_rotate_fp
    }
    \bool_lazy_and:nnT {
        \l_hawkdraw_shading_circular_bool
    } {
        \str_if_eq_p:nn {#1} { radial }
    } {
        \dim_set:Nn \l_hawkdraw_shading_bbox_xmax_dim {
            \dim_max:nn {
                \l_hawkdraw_shading_bbox_xmax_dim
            } {
                \l_hawkdraw_shading_bbox_ymax_dim
            }
        }
        \dim_set:Nn \l_hawkdraw_shading_bbox_xmin_dim {
            \dim_min:nn {
                \l_hawkdraw_shading_bbox_xmin_dim
            } {
                \l_hawkdraw_shading_bbox_ymin_dim
            }
        }
        \dim_set_eq:NN \l_hawkdraw_shading_bbox_ymax_dim
            \l_hawkdraw_shading_bbox_xmax_dim
        \dim_set_eq:NN \l_hawkdraw_shading_bbox_ymin_dim
            \l_hawkdraw_shading_bbox_xmin_dim
    }
    \box_resize_to_wd_and_ht:Nnn \l_hawkdraw_shading_box {
        \fp_to_dim:n {
            (
                \l_hawkdraw_shading_bbox_xmax_dim -
                \l_hawkdraw_shading_bbox_xmin_dim
            ) * 2
        }
    } {
        \fp_to_dim:n {
            (
                \l_hawkdraw_shading_bbox_ymax_dim -
                \l_hawkdraw_shading_bbox_ymin_dim
            ) * 2
        }
    }
    \draw_box_use:Nn \l_hawkdraw_shading_box {
        (
            \l_hawkdraw_shading_bbox_xmin_dim -
            \l_hawkdraw_shading_bbox_xmax_dim
        ,
            \l_hawkdraw_shading_bbox_ymin_dim -
            \l_hawkdraw_shading_bbox_ymax_dim
        )
    }
}

% ===

% We use a Coons patch mesh for conic shadings where multiple circular-segment shaped patches form a circle.
% Two corners of the patch are at the center of the circle, the other two at the outer border forming a curve.
% We use four patches for each color in order to be able to generate a circle (or at least an approximation)
% with only two colors. Therefore, we need to interpolate a few colors so we end up with 2n + 1 colors.

% In theory we can use patches with different flags and reduce the size of the stream, but in practice this
% means that the path direction of every second patch needs to be reversed. Since we don't need a lot of
% patches to create a conic shading, we just use flag-0 patches only.

% We need to output a binary stream. It is theoretically possinle to output a hex stream and set the relevant
% /Filter, but when PDFTeX compresses the stream, it simply appends a /Filter entry to the attributes of the
% PDF object without taking into account an already existing /Filter entry which results in faulty syntax.
% We need different approaches to output such a stream for PDFTeX and LuaTeX. Since dvipdfmx requires a hex
% stream which it converts to a binary stream, we first create a hex stream.

\cs_new_protected:Npn \hawkdraw_shading_conic_stream:N #1 {
    \tl_set:Ne #1 {
        \seq_map_indexed_function:NN
            \l__hawkdraw_shading_color_values_seq
            \__hawkdraw_shading_conic_stream_aux:nn
    }
}

\cs_new:Npn \__hawkdraw_shading_conic_stream_aux:nn #1#2 {
    % skip first entry in seq which is the start color
    \int_compare:nNnF {#1} = { \c_one_int } {
        % flag (4 bits)
        \hawkdraw_hex_to_stream:n { 00 }
        % patch is pie segment going from 0 deg to 360 / ( n * 4 ) deg
        % points (12 points, 2 * 16 bits each)
        \prg_replicate:nn { 2 } {
            \__hawkdraw_shading_conic_stream_point_polar:nn { 0 } {
                \l_hawkdraw_shading_angle_delta_fp /
                ( \seq_count:N \l__hawkdraw_shading_color_values_seq - 1 ) * ( #1 - 2 )
            }
        }
        \prg_replicate:nn { 2 } {
            \__hawkdraw_shading_conic_stream_point_polar:nn { 1 } {
                \l_hawkdraw_shading_angle_delta_fp /
                ( \seq_count:N \l__hawkdraw_shading_color_values_seq - 1 ) * ( #1 - 2 )
            }
        }
        \__hawkdraw_shading_conic_stream_point_polar:nn { 1 } {
            \l_hawkdraw_shading_angle_delta_fp /
            ( \seq_count:N \l__hawkdraw_shading_color_values_seq - 1 ) * ( #1 - 1.75 )
        }
        \__hawkdraw_shading_conic_stream_point_polar:nn { 1 } {
            \l_hawkdraw_shading_angle_delta_fp /
            ( \seq_count:N \l__hawkdraw_shading_color_values_seq - 1 ) * ( #1 - 1.25 )
        }
        \prg_replicate:nn { 2 } {
            \__hawkdraw_shading_conic_stream_point_polar:nn { 1 } {
                \l_hawkdraw_shading_angle_delta_fp /
                ( \seq_count:N \l__hawkdraw_shading_color_values_seq - 1 ) * ( #1 - 1 )
            }
        }
        \prg_replicate:nn { 3 } {
            \__hawkdraw_shading_conic_stream_point_polar:nn { 0 } {
                \l_hawkdraw_shading_angle_delta_fp /
                ( \seq_count:N \l__hawkdraw_shading_color_values_seq - 1 ) * ( #1 - 1 )
            }
        }
        \__hawkdraw_shading_conic_stream_point_polar:nn { 0 } {
            \l_hawkdraw_shading_angle_delta_fp /
            ( \seq_count:N \l__hawkdraw_shading_color_values_seq - 1 ) * ( #1 - 2 )
        }
        % colors (4 colors, 3 * 8 bits each for RGB)
        \hawkdraw_hex_to_stream:e {
            \hawkdraw_color_pdf_to_hex:e {
                \seq_item:Nn \l__hawkdraw_shading_color_values_seq { #1 - 1 }
            }
        }
        \hawkdraw_hex_to_stream:e {
            \hawkdraw_color_pdf_to_hex:e {
                \seq_item:Nn \l__hawkdraw_shading_color_values_seq { #1 - 1 }
            }
        }
        \hawkdraw_hex_to_stream:e {
            \hawkdraw_color_pdf_to_hex:n {#2}
        }
        \hawkdraw_hex_to_stream:e {
            \hawkdraw_color_pdf_to_hex:n {#2}
        }
    }
}

\cs_new:Npn \__hawkdraw_shading_conic_stream_point_polar:nn #1#2 {
    % get coordinates from shifted point polar and normalize to FFFF
    \hawkdraw_hex_to_stream:e {
        \int_format:nn {
            \fp_to_int:n {
                \hawkdraw_tuple_use_i:n {
                    ( 50bp , 50bp ) + (
